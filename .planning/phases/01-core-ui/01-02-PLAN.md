Replace the skeleton app.js with full task creation logic:

```js
// Task Tracker — main app logic

/** @type {{ id: number, title: string }[]} */
let tasks = [];
let nextId = 1;

const form = document.getElementById('task-form');
const input = document.getElementById('task-input');
const taskList = document.getElementById('task-list');

/**
 * Render the current tasks array into #task-list.
 * Clears the list and re-renders from scratch on each call.
 */
function renderTasks() {
  taskList.innerHTML = '';
  if (tasks.length === 0) {
    const empty = document.createElement('li');
    empty.textContent = 'No tasks yet. Add one above!';
    empty.style.color = '#999';
    empty.dataset.testid = 'empty-message';
    taskList.appendChild(empty);
    return;
  }
  tasks.forEach(function (task) {
    const li = document.createElement('li');
    li.textContent = task.title;
    li.dataset.taskId = task.id;
    taskList.appendChild(li);
  });
}

/**
 * Handle form submission: create a new task and re-render.
 * @param {Event} event
 */
form.addEventListener('submit', function (event) {
  event.preventDefault();
  const title = input.value.trim();
  if (!title) return;
  tasks.push({ id: nextId++, title: title });
  input.value = '';
  renderTasks();
});

// Initial render (shows empty state)
renderTasks();
```

Key behaviors to implement exactly as shown:
- `tasks` is a module-level array (Phase 2 will wrap this with localStorage)
- `renderTasks()` clears and re-renders the full list (not incremental DOM append)
- Empty state shows a friendly message with `data-testid="empty-message"` for Playwright
- Each task `<li>` has `data-task-id="{id}"` attribute for test targeting
- Input is cleared after submit (`input.value = ''`)
- `event.preventDefault()` prevents page reload on form submit

```js
const { test, expect } = require('@playwright/test');

test.describe('Task Tracker — Phase 1 Core UI', () => {

  test.beforeEach(async ({ page }) => {
    await page.goto('/');
  });

  test('page loads with heading and input visible', async ({ page }) => {
    await expect(page.locator('h1, header')).toContainText('Task Tracker');
    await expect(page.locator('#task-input')).toBeVisible();
    await expect(page.locator('button[type="submit"]')).toBeVisible();
  });

  test('shows empty state message on first load', async ({ page }) => {
    await expect(page.locator('[data-testid="empty-message"]')).toBeVisible();
    await expect(page.locator('[data-testid="empty-message"]')).toContainText('No tasks yet');
  });

  test('user can add a task and it appears in the list', async ({ page }) => {
    await page.locator('#task-input').fill('Buy groceries');
    await page.locator('button[type="submit"]').click();

    await expect(page.locator('#task-list li')).toContainText('Buy groceries');
    // Empty message should be gone
    await expect(page.locator('[data-testid="empty-message"]')).not.toBeVisible();
  });

  test('input field is cleared after task is added', async ({ page }) => {
    await page.locator('#task-input').fill('Walk the dog');
    await page.keyboard.press('Enter');

    await expect(page.locator('#task-input')).toHaveValue('');
  });

  test('user can add multiple tasks and all appear in the list', async ({ page }) => {
    const taskTitles = ['Task one', 'Task two', 'Task three'];

    for (const title of taskTitles) {
      await page.locator('#task-input').fill(title);
      await page.keyboard.press('Enter');
    }

    const items = page.locator('#task-list li');
    await expect(items).toHaveCount(3);
    await expect(items.nth(0)).toContainText('Task one');
    await expect(items.nth(1)).toContainText('Task two');
    await expect(items.nth(2)).toContainText('Task three');
  });

  test('submitting empty input does not add a task', async ({ page }) => {
    // Add one real task first so we can count
    await page.locator('#task-input').fill('Real task');
    await page.keyboard.press('Enter');
    await expect(page.locator('#task-list li')).toHaveCount(1);

    // Now try submitting empty — count should remain 1
    await page.locator('#task-input').fill('   ');
    await page.keyboard.press('Enter');
    await expect(page.locator('#task-list li')).toHaveCount(1);
  });

});
```
