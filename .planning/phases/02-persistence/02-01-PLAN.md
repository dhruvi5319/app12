Modify app.js to persist tasks to localStorage. Phase 1 built `app.js` with a module-level `let tasks = []`, a `renderTasks()` function, and a form submit handler. Phase 2 wraps that array with localStorage read/write.

**Replace the existing app.js** with this implementation:

```js
// Task Tracker — main app logic with localStorage persistence

const STORAGE_KEY = 'task-tracker-tasks';

/** @type {{ id: number, title: string }[]} */
let tasks = [];
let nextId = 1;

const form = document.getElementById('task-form');
const input = document.getElementById('task-input');
const taskList = document.getElementById('task-list');

/**
 * Save the current tasks array to localStorage.
 */
function saveTasks() {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(tasks));
}

/**
 * Load tasks from localStorage. If nothing is stored, tasks stays [].
 * Also rehydrates nextId so IDs never collide after reload.
 */
function loadTasks() {
  const stored = localStorage.getItem(STORAGE_KEY);
  if (stored) {
    try {
      tasks = JSON.parse(stored);
      // Ensure nextId is always higher than any existing task id
      if (tasks.length > 0) {
        nextId = Math.max(...tasks.map(function (t) { return t.id; })) + 1;
      }
    } catch (e) {
      // Corrupt data — reset cleanly
      tasks = [];
      localStorage.removeItem(STORAGE_KEY);
    }
  }
}

/**
 * Render the current tasks array into #task-list.
 * Clears the list and re-renders from scratch on each call.
 */
function renderTasks() {
  taskList.innerHTML = '';
  if (tasks.length === 0) {
    const empty = document.createElement('li');
    empty.textContent = 'No tasks yet. Add one above!';
    empty.style.color = '#999';
    empty.dataset.testid = 'empty-message';
    taskList.appendChild(empty);
    return;
  }
  tasks.forEach(function (task) {
    const li = document.createElement('li');
    li.textContent = task.title;
    li.dataset.taskId = task.id;
    taskList.appendChild(li);
  });
}

/**
 * Handle form submission: create a new task and re-render.
 * @param {Event} event
 */
form.addEventListener('submit', function (event) {
  event.preventDefault();
  const title = input.value.trim();
  if (!title) return;
  tasks.push({ id: nextId++, title: title });
  input.value = '';
  saveTasks();
  renderTasks();
});

// On startup: load persisted tasks, then render
loadTasks();
renderTasks();
```

Key changes from Phase 1:
- Added `STORAGE_KEY` constant — use `'task-tracker-tasks'` (consistent key avoids collisions with other apps)
- Added `saveTasks()` — called after every mutation (`tasks.push`), writes `JSON.stringify(tasks)` to `localStorage`
- Added `loadTasks()` — called once on startup, reads and parses stored tasks; also rehydrates `nextId` so IDs don't reset to 1 after reload
- `loadTasks()` wraps `JSON.parse` in try/catch — corrupt storage data resets cleanly rather than crashing
- Startup sequence changed from `renderTasks()` alone → `loadTasks(); renderTasks();`
- `renderTasks()`, `form.addEventListener`, and the task `{id, title}` shape are unchanged (preserves Phase 1 Playwright test compatibility)

  <name>Task 2: Write Playwright E2E tests for localStorage persistence</name>
  <files>e2e/persistence.spec.js</files>
  <action>
Create `e2e/persistence.spec.js` with Playwright tests that prove tasks survive page reloads and tab close/reopen.

**Important:** Playwright's `page.reload()` reloads the page within the same browser context, preserving localStorage. This correctly simulates a user pressing F5. The `page.goto('/')` in `beforeEach` also starts fresh from localStorage (same context), simulating reopening a tab.

```js
const { test, expect } = require('@playwright/test');

test.describe('Task Tracker — Phase 2 Persistence', () => {

  test.beforeEach(async ({ page }) => {
    // Start each test with a clean localStorage state
    await page.goto('/');
    await page.evaluate(() => localStorage.clear());
    await page.reload();
  });

  test('tasks persist after page reload', async ({ page }) => {
    // Add two tasks
    await page.locator('#task-input').fill('Buy groceries');
    await page.keyboard.press('Enter');
    await page.locator('#task-input').fill('Walk the dog');
    await page.keyboard.press('Enter');

    // Confirm they appear before reload
    await expect(page.locator('#task-list li')).toHaveCount(2);

    // Reload and confirm both tasks are still present
    await page.reload();
    await expect(page.locator('#task-list li')).toHaveCount(2);
    await expect(page.locator('#task-list li').nth(0)).toContainText('Buy groceries');
    await expect(page.locator('#task-list li').nth(1)).toContainText('Walk the dog');
  });

  test('empty state is correct on first load with no stored tasks', async ({ page }) => {
    // localStorage was cleared in beforeEach — should show empty state
    await expect(page.locator('[data-testid="empty-message"]')).toBeVisible();
    await expect(page.locator('[data-testid="empty-message"]')).toContainText('No tasks yet');
  });

  test('tasks survive closing and reopening the tab (simulate via new page in same context)', async ({ browser }) => {
    const context = await browser.newContext();
    const page1 = await context.newPage();

    // Set up with clean state
    await page1.goto('/');
    await page1.evaluate(() => localStorage.clear());
    await page1.reload();

    // Add a task
    await page1.locator('#task-input').fill('Persistent task');
    await page1.keyboard.press('Enter');
    await expect(page1.locator('#task-list li')).toContainText('Persistent task');

    // Close tab (close page1) and open a new tab in the same browser context
    await page1.close();
    const page2 = await context.newPage();
    await page2.goto('/');

    // Task should still be there — localStorage is shared within the browser context
    await expect(page2.locator('#task-list li')).toContainText('Persistent task');

    await context.close();
  });

  test('new tasks added after reload are also persisted', async ({ page }) => {
    // Add a task, reload, add another task, reload again
    await page.locator('#task-input').fill('First task');
    await page.keyboard.press('Enter');
    await page.reload();

    await page.locator('#task-input').fill('Second task');
    await page.keyboard.press('Enter');
    await page.reload();

    // Both tasks must survive the second reload
    await expect(page.locator('#task-list li')).toHaveCount(2);
    await expect(page.locator('#task-list li').nth(0)).toContainText('First task');
    await expect(page.locator('#task-list li').nth(1)).toContainText('Second task');
  });

  test('task IDs do not collide after reload (nextId rehydration)', async ({ page }) => {
    // Add a task, reload, add another task — both must have unique data-task-id values
    await page.locator('#task-input').fill('Task A');
    await page.keyboard.press('Enter');
    await page.reload();

    await page.locator('#task-input').fill('Task B');
    await page.keyboard.press('Enter');

    const items = page.locator('#task-list li[data-task-id]');
    await expect(items).toHaveCount(2);

    const id1 = await items.nth(0).getAttribute('data-task-id');
    const id2 = await items.nth(1).getAttribute('data-task-id');
    expect(id1).not.toBe(id2);
  });

});
```

Note on test isolation: `localStorage.clear()` in `beforeEach` ensures each test starts with a clean slate. This prevents state leaking between tests while still testing real localStorage behavior.
